\begin{abstract}
\textbf{Purpose.} ``Shadow Math'' is a projection-first formalism: observed lower-dimensional shadows constrain (and sometimes uniquely determine)
higher-dimensional structure through lifts and consistency. We present three core theorems:

\begin{enumerate}[label=(\Alph*)]
\item \textbf{Single-view non-injectivity:} a single 2D projection does not uniquely determine 3D.
\item \textbf{Multi-view lift:} with two calibrated non-degenerate views and correct correspondences, a 3D point is uniquely recovered by triangulation.
\item \textbf{Dimension certificate:} from pairwise distances, the Gram (double-centered) matrix eigenstructure certifies the minimal Euclidean embedding dimension.
\end{enumerate}

\textbf{Empirical evidence.} We embed \OCTA\ Proof Lab v1.3 (JSON) and summarize its audit-grade certificate:
a 3D world is non-planar while a 2D-in-3D control is planar under noise, using eigen-ratios, out-of-plane energy, and MDS stress.
\end{abstract}

\section{Shadow Math: Core Definitions}
% ============================================================

\begin{definition}[Shadow map]
Let $X$ be a latent space (e.g., $\mathbb{R}^D$, a manifold, or a metric space) and $Y$ an observation space.
A \emph{shadow map} is a function $\pi:X\to Y$ that is typically surjective and not injective.
\end{definition}

\begin{definition}[Fiber / ambiguity set]
For $y\in Y$, the \emph{fiber} is $\pi^{-1}(y)=\{x\in X:\pi(x)=y\}$.
When $\pi$ is not injective, fibers contain multiple points: reconstruction from a single shadow is ill-posed without extra constraints.
\end{definition}

\begin{definition}[Lift]
A \emph{lift} is a (possibly partial) right-inverse $\ell:Y\to X$ such that $\pi(\ell(y))=y$ whenever defined.
In practice, lifts are induced by multi-shadow consistency, rigidity constraints, priors, or optimization.
\end{definition}

\begin{definition}[Multi-shadow system]
A multi-shadow observation consists of maps $\pi_i:X\to Y_i$ for $i=1,\dots,m$.
A latent point $x\in X$ is observed as $(y_1,\dots,y_m)$ with $y_i=\pi_i(x)$.
A \emph{consistent lift} is an $x$ such that $x\in\bigcap_{i=1}^m \pi_i^{-1}(y_i)$.
\end{definition}

\begin{remark}[What ``higher dimension'' means here]
Throughout, ``higher dimension'' is \emph{mathematical}: the minimal Euclidean dimension needed to realize the constraints.
No physical claim is made about spacetime; we prove statements about geometric consistency and embedding dimension.
\end{remark}

% ============================================================
\section{Theorem A: Single-View Non-Injectivity}
% ============================================================

\subsection{Pinhole projection model}
Let $\pi_f:\mathbb{R}^3_{z>0}\to\mathbb{R}^2$ be calibrated pinhole projection with focal length $f>0$:
\[
  \pi_f(X,Y,Z) = \left(\frac{fX}{Z},\, \frac{fY}{Z}\right).
\]

\begin{theorem}[Ray fiber; no unique single-view lift]\label{thm:A}
For any $(u,v)\in\mathbb{R}^2$, the fiber $\pi_f^{-1}(u,v)$ contains infinitely many points (a 1D ray). Hence $\pi_f$ is not injective.
\end{theorem}

\begin{proof}
Fix $(u,v)$. The constraints $u=fX/Z$ and $v=fY/Z$ imply $X=(u/f)Z$ and $Y=(v/f)Z$ for any $Z>0$.
Therefore the entire ray $\{((u/f)Z,(v/f)Z,Z):Z>0\}$ maps to $(u,v)$, so the fiber is infinite and $\pi_f$ is not injective.
\end{proof}

\begin{corollary}[Non-uniqueness of reconstruction from one image]
Any algorithm that reconstructs 3D from a single 2D image must inject extra assumptions (priors, learned models, symmetry, rigidity, known scale, etc.).
\end{corollary}

% ============================================================
\section{Theorem B: Multi-View Lift via Triangulation}
% ============================================================

\subsection{Back-projected rays}
A calibrated camera defines $\pi_i:\mathbb{R}^3\to\mathbb{R}^2$ with optical center $C_i$.
For an observation $x_i\in\mathbb{R}^2$, the set of 3D points that project to $x_i$ is a ray (or line) in 3D:
\[
  L_i(x_i) = \{C_i + t\,d_i(x_i): t>0\},
\]
where $d_i(x_i)$ is the unit direction associated with pixel $x_i$.

\begin{lemma}[Two distinct lines intersect in at most one point]\label{lem:lines}
If $L_1$ and $L_2$ are distinct lines in $\mathbb{R}^3$, then $|L_1\cap L_2|\le 1$.
\end{lemma}

\begin{proof}
If two distinct lines intersect at two points, they must coincide (a line is determined by two points), contradiction.
\end{proof}

\begin{theorem}[Two-view triangulation (noise-free, correct correspondence)]\label{thm:B}
Let cameras $1$ and $2$ be calibrated with distinct optical centers $C_1\neq C_2$.
Let $(x_1,x_2)$ be a correspondence generated by a single finite 3D point $X$.
Then $X$ is uniquely determined and equals the unique intersection:
\[
  X = L_1(x_1)\cap L_2(x_2).
\]
\end{theorem}

\begin{proof}
Since $(x_1,x_2)$ comes from $X$, we have $X\in L_1(x_1)$ and $X\in L_2(x_2)$ so the intersection is nonempty.
Because $C_1\neq C_2$, the rays are distinct lines. By Lemma~\ref{lem:lines}, their intersection contains at most one point.
Therefore the intersection is exactly $\{X\}$.
\end{proof}

\begin{remark}[Noise and practical triangulation]
With noise, rays do not intersect exactly; one solves a least-squares approximation (e.g., DLT / SVD).
Three or more views improve stability by over-determining the system.
\end{remark}

\section{Shadow Calculus: Fiber Dimension from Jacobian Rank}
% ============================================================

\begin{theorem}[Constant Rank Theorem (informal statement)]\label{thm:rank}
Let $\pi:\mathbb{R}^D\to\mathbb{R}^d$ be $C^1$ and suppose $\mathrm{rank}\,D\pi(x)=r$ is constant in a neighborhood of $x$.
Then near $x$, the fiber $\pi^{-1}(\pi(x))$ is a $(D-r)$-dimensional submanifold.
\end{theorem}

\begin{remark}
For a generic smooth projection from $D$ to $d$ with full rank $r=d$, the fiber is locally $(D-d)$-dimensional.
For pinhole projection $\mathbb{R}^3\to\mathbb{R}^2$, this predicts a 1D ambiguity set (the ray), matching Theorem~\ref{thm:A}.
\end{remark}

% ============================================================
\section{Theorem C: Gram Eigenstructure Certifies Dimension}
% ============================================================

\subsection{Distances, centering, and Gram}
Given points $p_1,\dots,p_n\in\mathbb{R}^k$, define squared distances $D^{(2)}_{ij}=\|p_i-p_j\|_2^2$ and the centering matrix $J=I-\frac{1}{n}\mathbf{1}\mathbf{1}^\top$.
Define the double-centered Gram candidate
\[
  B = -\frac{1}{2} J D^{(2)} J.
\]

\begin{theorem}[Euclidean distance matrix equivalence]\label{thm:C}
If $D^{(2)}$ is generated by some points in $\mathbb{R}^k$, then $B\succeq 0$ and $\mathrm{rank}(B)\le k$.
Conversely, if $B\succeq 0$ and $\mathrm{rank}(B)=r$, then there exist points in $\mathbb{R}^r$ whose squared distances equal $D^{(2)}$.
\end{theorem}

\begin{proof}
(\emph{Forward.}) Let $P$ be the centered coordinate matrix (rows $p_i$ with mean zero). Then $G=PP^\top$ is positive semidefinite and $\mathrm{rank}(G)\le k$.
Classical MDS shows $G=-\frac12 JD^{(2)}J=B$.

(\emph{Reverse.}) If $B=U\Lambda U^\top$ with $\Lambda\succeq 0$, define $X=U_r\Lambda_r^{1/2}$ using the positive eigenpairs.
Then $XX^\top=B$ is a Gram matrix in $\mathbb{R}^r$ and reproduces $D^{(2)}$.
\end{proof}

\begin{corollary}[Certified lower bound on embedding dimension]\label{cor:C}
If $B$ has at least $k+1$ strictly positive eigenvalues, then the distances cannot embed isometrically in $\mathbb{R}^k$.
Thus the intrinsic Euclidean dimension is at least $k+1$.
\end{corollary}

\subsection{Noise-robust planarity certificate}
Let eigenvalues be $\lambda_1\ge\lambda_2\ge\lambda_3\ge\cdots$ and define $\lambda_i^+=\max(\lambda_i,0)$.

\begin{definition}[Out-of-plane ratios]
\[
  \rho \coloneqq \frac{\lambda_3^+}{\lambda_2^+}\ \ (\lambda_2^+>0),\qquad
  \eta \coloneqq \frac{\lambda_3^+}{\sum_i \lambda_i^+}.
\]
\end{definition}

\begin{definition}[Energy effective dimension]
Given $\varepsilon\in(0,1)$,
\[
  k_{\mathrm{eff}}(\varepsilon)=\min\left\{k:\frac{\sum_{i>k}\lambda_i^+}{\sum_i \lambda_i^+}\le\varepsilon\right\}.
\]
\end{definition}

\begin{definition}[MDS stress (squared)]
Let $\hat X^{(k)}$ be the classical MDS embedding using the top $k$ eigenpairs of $B$.
Let $D$ be original distances and $\hat D^{(k)}$ be distances in $\hat X^{(k)}$.
Define
\[
  \mathrm{stress}(k)=\left(\frac{\|D-\hat D^{(k)}\|_F}{\|D\|_F}\right)^2.
\]
\end{definition}

\begin{proposition}[Planar data implies $\lambda_3=0$ (noise-free)]\label{prop:planar}
If points lie in an affine 2D subspace, then $\mathrm{rank}(B)\le 2$ and $\lambda_3=0$, so $\rho=\eta=0$, $k_{\mathrm{eff}}=2$, and $\mathrm{stress}(2)=0$.
\end{proposition}

\begin{proof}
After centering, points lie in a 2D linear subspace, so the centered coordinate matrix has rank at most 2.
Thus $B=PP^\top$ has rank at most 2 and $\lambda_3=0$. Exact MDS in $k=2$ reproduces all distances so stress is 0.
\end{proof}

\begin{definition}[Audit-grade planarity rule]
Fix thresholds $(\rho_\tau,\eta_\tau)$ and declare \emph{planar} iff $\rho\le \rho_\tau$ and $\eta\le \eta_\tau$.
In Proof Lab v1.3 we use $(\rho_\tau,\eta_\tau)=(\RhoTau,\EtaTau)$.
\end{definition}

\section{Interpretation: What is Proven and What is Not}
% ============================================================

\subsection{What is proven (mathematically)}
\begin{itemize}
\item A single calibrated 2D pinhole projection is not injective; its fiber is a ray (Theorem~\ref{thm:A}).
\item Two calibrated views with non-zero baseline uniquely recover a 3D point in the noise-free consistent case (Theorem~\ref{thm:B}).
\item The Gram eigenstructure of distances certifies intrinsic Euclidean embedding dimension (Theorem~\ref{thm:C} and Corollary~\ref{cor:C}).
\item In noisy data, eigen-ratios ($\rho$), out-of-plane energy ($\eta$), and stress provide an audit-grade planarity classifier, validated by ablation (Proof Lab v1.3).
\end{itemize}

\subsection{What is not proven (physics)}
This document does not prove physical extra spacetime dimensions. It proves a rigorous statement of \emph{dimensional necessity for constraints}:
if a set of constraints (e.g., distances) is incompatible with any embedding in $\mathbb{R}^d$, then a higher-dimensional mathematical model is required.
